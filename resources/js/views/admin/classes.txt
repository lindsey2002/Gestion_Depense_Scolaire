<template>
  <div class="p-6 bg-gray-50 min-h-screen">
    <div class="max-w-6xl mx-auto space-y-6">
      
      <div class="flex justify-between items-center">
        <h1 class="text-3xl font-bold text-gray-800">Configuration des Classes</h1>
        <span class="bg-blue-100 text-blue-800 text-xs font-semibold px-2.5 py-0.5 rounded">Espace Admin</span>
      </div>

      <div v-if="error" class="bg-red-50 text-red-600 p-3 rounded-lg text-sm font-semibold">{{ error }}</div>
      <div v-if="success" class="bg-green-50 text-green-600 p-3 rounded-lg text-sm font-semibold">{{ success }}</div>

      <div class="grid grid-cols-1 lg:grid-cols-3 gap-6">
        <div class="bg-white p-6 rounded-xl shadow-sm border border-gray-100 h-fit">
          <h2 class="text-xl font-bold text-gray-700 mb-4">Ajouter une classe</h2>
          <form @submit.prevent="handleCreateClasse" class="space-y-4">
            <div>
              <label class="block text-sm font-medium text-gray-700">Associer à une Vague de rentrée</label>
              <select v-model="form.vague_id" required class="mt-1 block w-full p-2.5 border border-gray-300 rounded-lg bg-white shadow-sm focus:ring-blue-500 focus:border-blue-500">
                <option value="" disabled>Sélectionnez une vague</option>
                <option v-for="vague in vagues" :key="vague.id" :value="vague.id">
                  {{ vague.nom }} (Début : {{ vague.date_debut }})
                </option>
              </select>
            </div>
            <div>
              <label class="block text-sm font-medium text-gray-700">Nom de la classe</label>
              <input v-model="form.nom" type="text" placeholder="ex: Licence 3 Informatique" required class="mt-1 block w-full p-2.5 border border-gray-300 rounded-lg shadow-sm focus:ring-blue-500 focus:border-blue-500">
            </div>

            <div>
              <label class="block text-sm font-medium text-gray-700">Montant de la scolarité (FCFA)</label>
              <input v-model="form.montant_scolarite" type="number" placeholder="ex: 500000" required class="mt-1 block w-full p-2.5 border border-gray-300 rounded-lg shadow-sm focus:ring-blue-500 focus:border-blue-500">
            </div>

            <button type="submit" :disabled="loadingBtn" class="w-full bg-blue-600 text-white p-2.5 rounded-lg font-bold hover:bg-blue-700 transition duration-200">
              {{ loadingBtn ? 'Création...' : 'Enregistrer la classe' }}
            </button>
          </form>
        </div>

        <div class="lg:col-span-2 bg-white rounded-xl shadow-sm border border-gray-100 overflow-hidden">
          <div class="p-6 border-b border-gray-100">
            <h2 class="text-xl font-bold text-gray-700">Liste des classes existantes</h2>
          </div>

          <div v-if="loadingTable" class="p-6 text-center text-gray-500">Chargement des données...</div>

          <div v-else-if="classes.length === 0" class="p-6 text-center text-gray-500">Aucune classe configurée pour le moment.</div>

          <table v-else class="w-full text-left border-collapse">
            <thead>
              <tr class="bg-gray-50 text-gray-600 uppercase text-xs font-bold border-b border-gray-100">
                <th class="p-4">ID</th>
                <th class="p-4">Nom de la vague</th>
                <th class="p-4">Nom de la classe</th>
                <th class="p-4">Montant Scolarité</th>
                <th class="p-4 text-center">Actions</th>
              </tr>
            </thead>
            <tbody class="divide-y divide-gray-100 text-gray-700">
              <tr v-for="classe in classes" :key="classe.id" class="hover:bg-gray-50 transition">
                <td class="p-4 font-semibold text-gray-500">#{{ classe.id }}</td>
                <td class="p-4 font-semibold text-gray-600">
                  {{ classe.vague ? classe.vague.nom : 'Aucune vague' }}
                </td>
                <td class="p-4 font-bold text-gray-800">{{ classe.nom }}</td>
                <td class="p-4 font-semibold text-gray-700">
                    <div class="text-sm font-bold text-blue-600">{{ classe.tarif_mensuel }} FCFA / mois</div>
                    <div class="text-xs text-gray-500">Inscription : {{ classe.frais_inscription }} FCFA</div>
                </td>
                <td class="p-4 text-center">
                  <button @submit.prevent="" class="text-red-600 hover:text-red-900 font-bold text-sm mx-2">Supprimer</button>
                </td>
              </tr>
            </tbody>
          </table>
        </div>
      </div>

    </div>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue';
import axios from 'axios';

const classes = ref([]);
const vagues = ref([]);
const loadingTable = ref(false);
const loadingBtn = ref(false);
const error = ref(null);
const success = ref(null);

const form = ref({
    nom: '',
    niveau: 'licence 1', // Valeur par défaut
    diminutif: '',
    cursus: 'cs',        // Valeur par défaut
    vague_id: '',         // Valeur par défaut
    tarif_mensuel: '',
    frais_inscription: ''
});

// Récupérer la liste des classes avec le Token Bearer et le préfixe v1
const fetchClasses = async () => {
    try {
        loadingTable.value = true;
        const token = localStorage.getItem('auth_token');
        const response = await axios.get('/api/v1/classes', {
            headers: { Authorization: `Bearer ${token}` }
        });
        classes.value = response.data; 
    } catch (err) {
        console.error(err);
        error.value = "Impossible de charger la liste des classes.";
    } finally {
        loadingTable.value = false;
    }
};

const fetchVagues = async () => {
    try {
        const token = localStorage.getItem('auth_token');
        const response = await axios.get('/api/v1/vagues', {
            headers: { Authorization: `Bearer ${token}` }
        });
        vagues.value = response.data;
    } catch (err) {
        console.error("Erreur lors du chargement des vagues :", err);
    }
};

// On enregistre une nouvelle classe (Route POST /api/v1/classes)
const handleCreateClasse = async () => {
    try {
        loadingBtn.value = true;
        error.value = null;
        success.value = null;
        
        const token = localStorage.getItem('auth_token');
        await axios.post('/api/v1/classes', form.value, {
            headers: { Authorization: `Bearer ${token}` }
        });

        success.value = "La classe a été créée avec succès !";
        form.value.nom = '';
        form.value.diminutif = '';
        form.value.vague_id = '';
        form.value.tarif_mensuel = '';
        form.value.frais_inscription = '';
        
        // On rafraîchit la liste immédiatement
        await fetchClasses();
    } catch (err) {
        console.error(err);
        error.value = err.response?.data?.message || "Erreur lors de la création de la classe.";
    } finally {
        loadingBtn.value = false;
    }
};

// Charger les données au chargement de la page
onMounted(() => {
    fetchClasses();
    fetchVagues();
});
</script>
